# RAG over Kubernetes Docs — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a deployed, citation-backed RAG service over pinned Kubernetes docs with reproducible retrieval/generation evals and ablations.

**Architecture:** Markdown docs → heading-aware chunks → local BGE embeddings + BM25 → reciprocal rank fusion → cross-encoder rerank → free-tier LLM with citation-enforcing prompt → FastAPI + minimal web UI. Evals are a separate harness over a 60-question, human-certified gold set.

**Tech Stack:** Python 3.12, uv, FastAPI, LanceDB, rank_bm25, sentence-transformers, OpenAI-compatible free APIs (Groq/Gemini), pytest, Docker, GitHub Actions, HF Spaces.

**Spec:** `docs/superpowers/specs/2026-10-29-llm-portfolio-design.md`

## Global Constraints

- Python `>=3.12,<3.13`; manage with `uv`.
- Pinned docs commit: `47132eb5367cfdcba0a3b33a0b3ffc81185bd7e8` (branch `release-1.34`).
- Corpus root inside the tarball: `website-47132eb5367cfdcba0a3b33a0b3ffc81185bd7e8/content/en/docs`.
- Include only `concepts/`, `tasks/`, `tutorials/`, `reference/`; exclude `reference/generated/`, `contribute/`, `home/`, `setup/`.
- Chunk target default 500 tokens, 15% overlap; never split a fenced code block.
- No API key may be required for unit tests or CI.
- No published number may be hand-written; it must come from a committed results JSON.
- No code copied from unlicensed repos. Adapted code gets an entry in `ATTRIBUTIONS.md`.
- Commit after every task.

## Review Focus

- **Question longer than 500 chars or empty:** API returns 422, never a 500.
- **LLM returns citations that do not exist (`[9]` with 5 chunks):** dropped, not rendered as a broken link.
- **Retrieved context is insufficient:** system refuses instead of hallucinating; refusal is scored.
- **Docs tarball contains a path-traversal entry:** extraction rejects it.
- **Docker/Spaces has no API key:** `/health` works and `/ask` returns a clear 503/offline answer, not a stack trace.
- **Code fence contains blank lines:** chunker keeps it intact.
- **Gold set has duplicate IDs or wrong split counts:** validator fails before an eval runs.

            index_dir=os.getenv("RAG_INDEX_DIR", "data/index"),
            llm_provider=provider,
            llm_model=os.getenv("LLM_MODEL", defaults["model"]),
            llm_base_url=os.getenv("LLM_BASE_URL", defaults["base_url"]),
            llm_api_key=os.getenv(defaults["key_env"]),
            offline=os.getenv("RAG_OFFLINE", "0") == "1",

# src/rag/ingest.py
from __future__ import annotations

import io
import tarfile
from collections.abc import Iterator
from pathlib import Path

import httpx

from rag.config import Settings

INCLUDE_PREFIXES = ("concepts/", "tasks/", "tutorials/", "reference/")
EXCLUDE_PREFIXES = ("reference/generated/",)


def _is_safe_member(name: str, dest: Path) -> bool:
    target = (dest / name).resolve()
    return str(target).startswith(str(dest.resolve()) + "/")


def extract_tarball(archive: Path, dest: Path) -> Path:
    dest.mkdir(parents=True, exist_ok=True)
    with tarfile.open(archive, "r:gz") as tf:
        for member in tf.getmembers():
            if not _is_safe_member(member.name, dest):
                raise ValueError(f"unsafe path in archive: {member.name}")
        tf.extractall(dest, filter="data")
    return dest


def download_and_extract(settings: Settings) -> Path:
    dest = Path(settings.docs_dir)
    marker = dest / ".complete"
    if marker.exists():
        return dest
    dest.mkdir(parents=True, exist_ok=True)
    archive = dest / "docs.tar.gz"
    with httpx.stream("GET", settings.docs_url, follow_redirects=True, timeout=120) as r:
        r.raise_for_status()
        with archive.open("wb") as f:
            for block in r.iter_bytes():
                f.write(block)
    extract_tarball(archive, dest)
    (dest / settings.docs_prefix).rename(dest / "content")
    marker.write_text("ok")
    archive.unlink(missing_ok=True)
    return dest


def iter_markdown(docs_root: Path) -> Iterator[Path]:
    content = docs_root / "content"
    for path in sorted(content.rglob("*.md")):
        rel = path.relative_to(content).as_posix()
        if not rel.startswith(INCLUDE_PREFIXES):
            continue
        if rel.startswith(EXCLUDE_PREFIXES):
            continue
        yield path
```

````python
# src/rag/chunk.py
from __future__ import annotations

import hashlib
import re
from collections.abc import Callable

from rag.models import Chunk

FRONTMATTER_RE = re.compile(r"\A---\s*\n(.*?)\n---\s*\n", re.S)
SHORTCODE_RE = re.compile(r"\{\{[<%].*?[>%]\}\}", re.S)
HEADING_RE = re.compile(r"^(#{1,6})\s+(.+?)\s*$", re.M)


def _frontmatter(text: str) -> tuple[dict[str, str], str]:
    match = FRONTMATTER_RE.match(text)
    if not match:
        return {}, text
    meta: dict[str, str] = {}
    for line in match.group(1).splitlines():
        if ":" in line:
            key, value = line.split(":", 1)
            meta[key.strip()] = value.strip().strip('"').strip("'")
    return meta, text[match.end():]


def _slug(heading: str) -> str:
    slug = re.sub(r"[^a-z0-9\s-]", "", heading.lower())
    return re.sub(r"\s+", "-", slug.strip())


def _url(source_path: str, anchor_heading: str) -> str:
    rel = source_path.removesuffix(".md")
    rel = rel.replace("/_index", "").strip("/")
    base = f"https://kubernetes.io/docs/{rel}/"
    return f"{base}#{_slug(anchor_heading)}" if anchor_heading else base


def _split_blocks(text: str) -> list[str]:
    blocks: list[str] = []
    buf: list[str] = []
    in_fence = False
    for line in text.splitlines():
        if line.startswith("```"):
            in_fence = not in_fence
        if not in_fence and not line.strip():
            if buf:
                blocks.append("\n".join(buf).strip())
                buf = []
        else:
            buf.append(line)
    if buf:
        blocks.append("\n".join(buf).strip())
    return [b for b in blocks if b]


def chunk_text(
    text: str,
    max_tokens: int,
    overlap: float,
    count_tokens: Callable[[str], int],
) -> list[str]:
    blocks = _split_blocks(text)
    chunks: list[str] = []
    current: list[str] = []
    current_tokens = 0
    overlap_tokens = int(max_tokens * overlap)

    def flush() -> None:
        nonlocal current, current_tokens
        if current:
            chunks.append("\n\n".join(current))
        tail: list[str] = []
        tail_tokens = 0
        for block in reversed(current):
            if tail_tokens >= overlap_tokens:
                break
            tail.insert(0, block)
            tail_tokens += count_tokens(block)
        current = tail
        current_tokens = tail_tokens

    for block in blocks:
        tokens = count_tokens(block)
        if current and current_tokens + tokens > max_tokens:
            flush()
        current.append(block)
        current_tokens += tokens

    if current:
        chunks.append("\n\n".join(current))
    return [c for c in chunks if c.strip()]


def parse_markdown(
    text: str,
    source_path: str,
    version: str,
    chunk_tokens: int = 500,
    overlap: float = 0.15,
    count_tokens: Callable[[str], int] | None = None,
) -> list[Chunk]:
    count_tokens = count_tokens or (lambda s: len(s.split()))
    meta, body = _frontmatter(text)
    if meta.get("draft", "false").lower() == "true":
        return []
    body = SHORTCODE_RE.sub("", body)

    chunks: list[Chunk] = []
    heading_stack: list[tuple[int, str]] = []
    positions = list(HEADING_RE.finditer(body))
    sections: list[tuple[str, str, str]] = []
    if not positions:
        sections.append(("", meta.get("title", ""), body))
    for i, match in enumerate(positions):
        level = len(match.group(1))
        title = match.group(2).strip()
        while heading_stack and heading_stack[-1][0] >= level:
            heading_stack.pop()
        heading_stack.append((level, title))
        start = match.end()
        end = positions[i + 1].start() if i + 1 < len(positions) else len(body)
        heading_path = " > ".join(t for _, t in heading_stack)
        sections.append((heading_path, title, body[start:end].strip()))

    for heading_path, leaf_title, section in sections:
        if not section:
            continue
        for piece in chunk_text(section, chunk_tokens, overlap, count_tokens):
            digest = hashlib.sha256(f"{source_path}:{heading_path}:{piece}".encode()).hexdigest()[:16]
            chunks.append(
                Chunk(
                    id=digest,
                    text=piece,
                    url=_url(source_path, leaf_title),
                    heading=heading_path or meta.get("title", ""),
                    source_path=source_path,
                    version=version,
                )
            )
    return chunks
````

# src/rag/index.py
from __future__ import annotations

import json
import pickle
from pathlib import Path

import lancedb
import numpy as np
from rank_bm25 import BM25Okapi

from rag.config import Settings
from rag.models import Chunk


def _tokenize(text: str) -> list[str]:
    return [t for t in text.lower().split() if t]


class Index:
    def __init__(self, table, bm25: BM25Okapi, ids: list[str], chunks: dict[str, Chunk], settings: Settings):
        self.table = table
        self.bm25 = bm25
        self.ids = ids
        self.chunks = chunks
        self.settings = settings

    @property
    def count(self) -> int:
        return len(self.ids)

    def get(self, chunk_id: str) -> Chunk:
        return self.chunks[chunk_id]

    def dense_search(self, vector: np.ndarray, k: int) -> list[tuple[str, float]]:
        rows = self.table.search(vector.tolist()).limit(k).to_list()
        return [(row["id"], float(row.get("_distance", 0.0))) for row in rows]

    def bm25_search(self, query: str, k: int) -> list[tuple[str, float]]:
        scores = self.bm25.get_scores(_tokenize(query))
        order = np.argsort(scores)[::-1][:k]
        return [(self.ids[i], float(scores[i])) for i in order]

    def close(self) -> None:
        self.table = None

    @classmethod
    def load(cls, settings: Settings) -> "Index":
        index_dir = Path(settings.index_dir)
        db = lancedb.connect(str(index_dir / "lancedb"))
        table = db.open_table("chunks")
        with (index_dir / "bm25.pkl").open("rb") as f:
            payload = pickle.load(f)
        rows = table.to_arrow().to_pylist()
        chunks = {
            row["id"]: Chunk(
                id=row["id"],
                text=row["text"],
                url=row["url"],
                heading=row["heading"],
                source_path=row["source_path"],
                version=row["version"],
            )
            for row in rows
        }
        return cls(table, payload["bm25"], payload["ids"], chunks, settings)


def build_index(chunks: list[Chunk], embedder, settings: Settings) -> Index:
    index_dir = Path(settings.index_dir)
    index_dir.mkdir(parents=True, exist_ok=True)
    vectors = embedder.encode([c.text for c in chunks])
    rows = [
        {
            "id": c.id,
            "text": c.text,
            "url": c.url,
            "heading": c.heading,
            "source_path": c.source_path,
            "version": c.version,
            "vector": vec.tolist(),
        }
        for c, vec in zip(chunks, vectors)
    ]
    db = lancedb.connect(str(index_dir / "lancedb"))
    table = db.create_table("chunks", data=rows, mode="overwrite")

    ids = [c.id for c in chunks]
    bm25 = BM25Okapi([_tokenize(c.text) for c in chunks])
    with (index_dir / "bm25.pkl").open("wb") as f:
        pickle.dump({"ids": ids, "bm25": bm25}, f)

    meta = {
        "chunks": len(chunks),
        "embed_model": settings.embed_model,
        "docs_url": settings.docs_url,
        "chunk_tokens": settings.chunk_tokens,
        "chunk_overlap": settings.chunk_overlap,
    }
    (index_dir / "meta.json").write_text(json.dumps(meta, indent=2))
    return Index(table, bm25, ids, {c.id: c for c in chunks}, settings)
```

# src/rag/retrieve.py
from __future__ import annotations

import time
from collections import defaultdict

from rag.models import RetrievedChunk


def reciprocal_rank_fusion(rankings: list[list[str]], k: int = 60) -> list[tuple[str, float]]:
    scores: dict[str, float] = defaultdict(float)
    for ranking in rankings:
        for rank, doc_id in enumerate(ranking, start=1):
            scores[doc_id] += 1.0 / (k + rank)
    return sorted(scores.items(), key=lambda item: (-item[1], item[0]))


class CrossEncoderReranker:
    def __init__(self, model_name: str) -> None:
        from sentence_transformers import CrossEncoder

        self.model = CrossEncoder(model_name)

    def score(self, query: str, texts: list[str]) -> list[float]:
        if not texts:
            return []
        return [float(s) for s in self.model.predict([(query, t) for t in texts])]


class Retriever:
    def __init__(
        self,
        index,
        embedder,
        reranker,
        settings=None,
        top_k_context: int = 5,
        dense_only: bool = False,
        use_reranker: bool = True,
    ):
        self.index = index
        self.embedder = embedder
        self.reranker = reranker
        self.dense_only = dense_only
        self.use_reranker = use_reranker and not dense_only
        self.top_k_dense = getattr(settings, "top_k_dense", 50)
        self.top_k_bm25 = getattr(settings, "top_k_bm25", 50)
        self.top_k_rerank = getattr(settings, "top_k_rerank", 20)
        self.rrf_k = getattr(settings, "rrf_k", 60)
        self.top_k_context = top_k_context
        self.last_timings: dict[str, int] = {}

    def retrieve(self, question: str) -> list[RetrievedChunk]:
        dense_started = time.perf_counter()
        vector = self.embedder.encode([question])[0]
        dense = self.index.dense_search(vector, self.top_k_dense)
        dense_ms = int((time.perf_counter() - dense_started) * 1000)
        dense_rank = {doc_id: i for i, (doc_id, _) in enumerate(dense, start=1)}
        if self.dense_only:
            self.last_timings = {"dense_ms": dense_ms, "bm25_ms": 0, "rerank_ms": 0}
            return [
                RetrievedChunk(
                    chunk=self.index.get(doc_id),
                    score=score,
                    dense_rank=rank,
                )
                for rank, (doc_id, score) in enumerate(dense[: self.top_k_context], start=1)
            ]

        bm25_started = time.perf_counter()
        bm25 = self.index.bm25_search(question, self.top_k_bm25)
        bm25_ms = int((time.perf_counter() - bm25_started) * 1000)
        bm25_rank = {doc_id: i for i, (doc_id, _) in enumerate(bm25, start=1)}
        fused = reciprocal_rank_fusion(
            [[doc_id for doc_id, _ in dense], [doc_id for doc_id, _ in bm25]],
            self.rrf_k,
        )
        if not self.use_reranker:
            self.last_timings = {"dense_ms": dense_ms, "bm25_ms": bm25_ms, "rerank_ms": 0}
            return [
                RetrievedChunk(
                    chunk=self.index.get(doc_id),
                    score=score,
                    dense_rank=dense_rank.get(doc_id),
                    bm25_rank=bm25_rank.get(doc_id),
                )
                for doc_id, score in fused[: self.top_k_context]
            ]

        candidates = [self.index.get(doc_id) for doc_id, _ in fused[: self.top_k_rerank]]
        rerank_started = time.perf_counter()
        scores = self.reranker.score(question, [c.text for c in candidates])
        rerank_ms = int((time.perf_counter() - rerank_started) * 1000)
        self.last_timings = {"dense_ms": dense_ms, "bm25_ms": bm25_ms, "rerank_ms": rerank_ms}
        ranked = sorted(zip(candidates, scores), key=lambda pair: -pair[1])[: self.top_k_context]
        return [
            RetrievedChunk(
                chunk=chunk,
                score=score,
                dense_rank=dense_rank.get(chunk.id),
                bm25_rank=bm25_rank.get(chunk.id),
                rerank_score=score,
            )
            for chunk, score in ranked
        ]
```

# src/rag/generate.py
from __future__ import annotations

import re
import time

from rag.config import Settings
from rag.models import Answer, Chunk, Citation

SYSTEM_PROMPT = """You answer questions about Kubernetes using ONLY the provided documentation excerpts.
Rules:
- Cite every factual claim with [n] matching the excerpt number.
- If the excerpts do not contain the answer, reply exactly: "I don't know based on the provided documentation."
- Do not use outside knowledge.
- Be concise: at most 180 words."""

CITATION_RE = re.compile(r"\[(\d+)\]")


def build_messages(question: str, chunks: list[Chunk]) -> list[dict[str, str]]:
    context = "\n\n".join(
        f"[{i}] {c.url}\nHeading: {c.heading}\n{c.text}" for i, c in enumerate(chunks, start=1)
    )
    return [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "user", "content": f"Documentation excerpts:\n\n{context}\n\nQuestion: {question}"},
    ]


class LLMClient:
    def __init__(self, settings: Settings) -> None:
        if not settings.llm_api_key:
            raise ValueError("LLM API key is not configured")
        from openai import OpenAI

        self.settings = settings
        self.client = OpenAI(base_url=settings.llm_base_url, api_key=settings.llm_api_key)

    def complete(self, messages: list[dict[str, str]], max_tokens: int) -> tuple[str, dict[str, int]]:
        response = self.client.chat.completions.create(
            model=self.settings.llm_model,
            messages=messages,
            temperature=0.0,
            max_tokens=max_tokens,
        )
        text = response.choices[0].message.content or ""
        usage = {
            "prompt_tokens": response.usage.prompt_tokens if response.usage else 0,
            "completion_tokens": response.usage.completion_tokens if response.usage else 0,
        }
        return text.strip(), usage


def _citations(text: str, chunks: list[Chunk]) -> list[Citation]:
    seen: set[int] = set()
    result: list[Citation] = []
    for raw in CITATION_RE.findall(text):
        index = int(raw)
        if index in seen or not 1 <= index <= len(chunks):
            continue
        seen.add(index)
        chunk = chunks[index - 1]
        result.append(Citation(index, chunk.url, chunk.heading, chunk.id))
    return sorted(result, key=lambda c: c.id)


def answer_question(question: str, chunks: list[Chunk], llm, settings: Settings) -> Answer:
    if settings.offline:
        return extractive_answer(question, chunks)
    started = time.monotonic()
    text, usage = llm.complete(build_messages(question, chunks), settings.max_answer_tokens)
    return Answer(
        text=text,
        citations=_citations(text, chunks),
        usage=usage,
        latency_ms=int((time.monotonic() - started) * 1000),
    )


def extractive_answer(question: str, chunks: list[Chunk]) -> Answer:
    if not chunks:
        return Answer(
            text="I don't know based on the provided documentation.",
            citations=[],
            usage={"prompt_tokens": 0, "completion_tokens": 0},
            latency_ms=0,
        )
    terms = {t.lower() for t in re.findall(r"[a-z0-9]+", question)}
    best = max(
        chunks,
        key=lambda c: len(terms & {t.lower() for t in re.findall(r"[a-z0-9]+", c.text)}),
    )
    sentences = re.split(r"(?<=[.!?])\s+", best.text.strip())
    text = " ".join(sentences[:2]).strip() + " [1]"
    return Answer(
        text=text,
        citations=[Citation(1, best.url, best.heading, best.id)],
        usage={"prompt_tokens": 0, "completion_tokens": 0},
        latency_ms=0,
    )
```

class CitationOut(BaseModel):

    heading: str

    citations: list[CitationOut]

def create_app(settings: Settings, retriever=None, llm=None) -> FastAPI:
    app = FastAPI(title="Kubernetes Docs RAG")

    app.state.retriever = retriever

    if STATIC_DIR.exists():
        app.mount("/static", StaticFiles(directory=STATIC_DIR), name="static")

    @app.get("/")
    def root():

    @app.get("/health")
    def health():
        return {
            "status": "ok" if app.state.error is None else "degraded",
            "chunks": getattr(app.state.retriever, "index", None).count
            if app.state.retriever is not None and hasattr(app.state.retriever, "index")
            else 0,
            "error": app.state.error,
        }

    @app.post("/ask", response_model=AskResponse)
    def ask(request: AskRequest):
        if app.state.retriever is None or (app.state.llm is None and not settings.offline):
            raise HTTPException(status_code=503, detail=f"service not ready: {app.state.error}")
        started = time.monotonic()
        retrieved = app.state.retriever.retrieve(request.question)
        if not retrieved:
            raise HTTPException(status_code=404, detail="no documents retrieved")
        chunks = [r.chunk for r in retrieved]
        answer: Answer = answer_question(request.question, chunks, app.state.llm, settings)
        return AskResponse(
            answer=answer.text,
            citations=[CitationOut(**c.__dict__) for c in answer.citations],
            retrieval={
                "chunks_used": len(chunks),
                "top_chunk_id": chunks[0].id,
                **getattr(app.state.retriever, "last_timings", {}),
            },
            usage=answer.usage,
            latency_ms=int((time.monotonic() - started) * 1000),
        )

    return app

# evals/schema.py
from __future__ import annotations

import json
from dataclasses import dataclass
from pathlib import Path

CATEGORIES = {"single-hop", "multi-hop", "troubleshooting", "unanswerable"}
EXPECTED = {"single-hop": 20, "multi-hop": 15, "troubleshooting": 15, "unanswerable": 10}


@dataclass(frozen=True)
class GoldSource:
    url: str
    heading: str


@dataclass(frozen=True)
class EvalQuestion:
    id: str
    split: str
    category: str
    question: str
    reference_answer: str
    gold_sources: list[GoldSource]
    required_facts: list[str]


def load_questions(path: str | Path) -> list[EvalQuestion]:
    rows = []
    for line in Path(path).read_text().splitlines():
        if not line.strip():
            continue
        raw = json.loads(line)
        rows.append(
            EvalQuestion(
                id=raw["id"],
                split=raw["split"],
                category=raw["category"],
                question=raw["question"],
                reference_answer=raw["reference_answer"],
                gold_sources=[GoldSource(**s) for s in raw["gold_sources"]],
                required_facts=raw["required_facts"],
            )
        )
    return rows


def validate_questions(questions: list[EvalQuestion]) -> list[str]:
    errors = []
    ids = [q.id for q in questions]
    if len(ids) != len(set(ids)):
        errors.append("duplicate question ids")
    for split, count in (("dev", 30), ("test", 30)):
        actual = sum(q.split == split for q in questions)
        if actual != count:
            errors.append(f"{split}: expected {count}, got {actual}")
    for category, count in EXPECTED.items():
        actual = sum(q.category == category for q in questions)
        if actual != count:
            errors.append(f"{category}: expected {count}, got {actual}")
    for q in questions:
        if q.category not in CATEGORIES:
            errors.append(f"{q.id}: bad category {q.category}")
        if q.category != "unanswerable" and not q.gold_sources:
            errors.append(f"{q.id}: missing gold sources")
        if q.category != "unanswerable" and not q.required_facts:
            errors.append(f"{q.id}: missing required facts")
    return errors

# evals/metrics.py
from __future__ import annotations

import math
import re


def _url_key(url: str) -> str:
    return url.split("#", 1)[0].rstrip("/")


def recall_at_k(retrieved: list[str], gold: set[str], k: int) -> float:
    gold_keys = {_url_key(g) for g in gold}
    if not gold_keys:
        return 1.0
    return sum(_url_key(url) in gold_keys for url in retrieved[:k]) / len(gold_keys)


def mrr_at_k(retrieved: list[str], gold: set[str], k: int) -> float:
    gold_keys = {_url_key(g) for g in gold}
    for rank, url in enumerate(retrieved[:k], start=1):
        if _url_key(url) in gold_keys:
            return 1.0 / rank
    return 0.0


def ndcg_at_k(retrieved: list[str], gold: set[str], k: int) -> float:
    gold_keys = {_url_key(g) for g in gold}
    dcg = sum(1.0 / math.log2(i + 2) for i, url in enumerate(retrieved[:k]) if _url_key(url) in gold_keys)
    idcg = sum(1.0 / math.log2(i + 2) for i in range(min(len(gold_keys), k)))
    return dcg / idcg if idcg else 0.0


def citation_precision_recall(cited: list[str], gold: set[str]) -> tuple[float, float]:
    if not cited:
        return 0.0, 0.0
    gold_keys = {_url_key(g) for g in gold}
    hits = sum(_url_key(url) in gold_keys for url in cited)
    return hits / len(cited), hits / len(gold_keys) if gold_keys else 0.0


def _normalize(text: str) -> str:
    return re.sub(r"\s+", " ", re.sub(r"[^a-z0-9 ]", " ", text.lower())).strip()


def fact_coverage(answer: str, facts: list[str]) -> float:
    if not facts:
        return 1.0
    normalized = _normalize(answer)
    return sum(_normalize(f) in normalized for f in facts) / len(facts)


def cohen_kappa(a: list[int], b: list[int]) -> float:
    if len(a) != len(b) or not a:
        raise ValueError("label lists must be non-empty and equal length")
    n = len(a)
    observed = sum(x == y for x, y in zip(a, b)) / n
    pa = sum(a) / n
    pb = sum(b) / n
    expected = pa * pb + (1 - pa) * (1 - pb)
    if expected == 1.0:
        return 1.0
    return (observed - expected) / (1 - expected)
```

# evals/judge.py
from __future__ import annotations

import hashlib
import json
from pathlib import Path

RUBRIC_VERSION = "v1"
JUDGE_PROMPT = """You are grading a RAG answer.
Return ONLY JSON with integer 0/1 fields:
- correct: does the answer cover the reference facts?
- faithful: is every claim supported by the context?
- citation_support: do the cited context items support the claims?
- refusal_correct: 1 if the question is unanswerable and the answer refuses, 0 if it hallucinates, null otherwise.
Also include a one-sentence rationale."""


def cache_key(question: str, candidate: str, reference: str, context: str) -> str:
    payload = f"{RUBRIC_VERSION}\n{question}\n{candidate}\n{reference}\n{context}"
    return hashlib.sha256(payload.encode()).hexdigest()


class Judge:
    def __init__(self, llm, cache_dir: str | Path = "evals/cache") -> None:
        self.llm = llm
        self.cache_dir = Path(cache_dir)
        self.cache_dir.mkdir(parents=True, exist_ok=True)

    def score(self, question: str, candidate: str, reference: str, context: str) -> dict:
        key = cache_key(question, candidate, reference, context)
        path = self.cache_dir / f"{key}.json"
        if path.exists():
            return json.loads(path.read_text())
        messages = [
            {"role": "system", "content": JUDGE_PROMPT},
            {
                "role": "user",
                "content": (
                    f"Question:\n{question}\n\nReference answer:\n{reference}\n\n"
                    f"Candidate answer:\n{candidate}\n\nContext:\n{context}"
                ),
            },
        ]
        text, _ = self.llm.complete(messages, max_tokens=200)
        start, end = text.find("{"), text.rfind("}")
        result = json.loads(text[start : end + 1])
        path.write_text(json.dumps(result, indent=2))
        return result
```

# evals/run_eval.py
from __future__ import annotations

import argparse
import json
import sys
import time
from dataclasses import replace
from datetime import datetime, timezone
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))
sys.path.insert(0, str(ROOT / "src"))

from evals.metrics import (
    citation_precision_recall,
    fact_coverage,
    mrr_at_k,
    ndcg_at_k,
    recall_at_k,
)
from evals.schema import EvalQuestion, load_questions


def run_eval(
    questions: list[EvalQuestion],
    retriever,
    answer_fn,
    judge=None,
) -> dict:
    rows = []
    for q in questions:
        retrieved = retriever.retrieve(q.question)
        urls = [r.chunk.url for r in retrieved]
        gold = {s.url for s in q.gold_sources}
        row = {
            "id": q.id,
            "category": q.category,
            "recall@5": recall_at_k(urls, gold, 5),
            "recall@10": recall_at_k(urls, gold, 10),
            "mrr@10": mrr_at_k(urls, gold, 10),
            "ndcg@10": ndcg_at_k(urls, gold, 10),
        }
        if answer_fn is not None:
            started = time.monotonic()
            answer = answer_fn(q.question, [r.chunk for r in retrieved])
            row["latency_ms"] = int((time.monotonic() - started) * 1000)
            row["answer"] = answer.text
            row["citations"] = [c.url for c in answer.citations]
            row["fact_coverage"] = fact_coverage(answer.text, q.required_facts)
            precision, recall = citation_precision_recall(row["citations"], gold)
            row["citation_precision"] = precision
            row["citation_recall"] = recall
            if judge is not None:
                context = "\n\n".join(r.chunk.text for r in retrieved)
                row["judge"] = judge.score(q.question, answer.text, q.reference_answer, context)
        rows.append(row)

    def mean(key: str) -> float:
        values = [r[key] for r in rows if key in r]
        return sum(values) / len(values) if values else 0.0

    return {
        "generated_at": datetime.now(timezone.utc).isoformat(),
        "questions": len(rows),
        "metrics": {
            key: mean(key)
            for key in row_keys(rows)
        },
        "rows": rows,
    }


def row_keys(rows: list[dict]) -> list[str]:
    keys: list[str] = []
    for row in rows:
        for key in row:
            if key not in keys and key not in {"id", "category", "answer", "citations", "judge"}:
                keys.append(key)
    return keys


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--questions", default="evals/questions.jsonl")
    parser.add_argument("--split", choices=["dev", "test"], default="dev")
    parser.add_argument("--limit", type=int, default=0)
    parser.add_argument("--smoke", action="store_true")
    parser.add_argument("--retrieval-only", action="store_true")
    parser.add_argument("--offline", action="store_true")
    parser.add_argument("--index-dir", default="")
    parser.add_argument("--retrieval-mode", choices=["dense", "hybrid", "hybrid_rerank"], default="hybrid_rerank")
    parser.add_argument("--output", default="")
    args = parser.parse_args()

    questions = [q for q in load_questions(args.questions) if q.split == args.split]
    if args.limit:
        questions = questions[: args.limit]
    if args.smoke:
        questions = questions[:15]

    from rag.config import Settings
    from rag.embed import Embedder
    from rag.index import Index
    from rag.retrieve import CrossEncoderReranker, Retriever

    settings = Settings.from_env()
    if args.offline:
        settings = replace(settings, offline=True)
    if args.index_dir:
        settings = replace(settings, index_dir=args.index_dir)
    retriever = Retriever(
        Index.load(settings),
        Embedder(settings.embed_model),
        CrossEncoderReranker(settings.rerank_model),
        settings,
        dense_only=args.retrieval_mode == "dense",
        use_reranker=args.retrieval_mode == "hybrid_rerank",
    )
    answer_fn = None
    judge = None
    if not args.retrieval_only:
        from evals.judge import Judge
        from rag.generate import LLMClient, answer_question

        llm = LLMClient(settings)
        answer_fn = lambda question, chunks: answer_question(question, chunks, llm, settings)
        judge = Judge(llm)

    result = run_eval(questions, retriever, answer_fn, judge)
    output = Path(args.output or f"evals/results/{args.split}-{int(time.time())}.json")
    output.parent.mkdir(parents=True, exist_ok=True)
    output.write_text(json.dumps(result, indent=2))
    print(json.dumps(result["metrics"], indent=2))
    print(f"wrote {output}")


if __name__ == "__main__":
    main()

    human = [int(r["human_correct"]) for r in rows]
    judge = [int(r["judge"]["correct"]) for r in rows]
    kappa = cohen_kappa(human, judge)
    print(f"n={len(rows)} agreement={sum(a == b for a, b in zip(human, judge)) / len(rows):.3f} kappa={kappa:.3f}")
    if kappa < 0.75:
        raise SystemExit("judge-human kappa below 0.75; revise the rubric before publishing")


if __name__ == "__main__":
    main()

# evals/summarize.py
from __future__ import annotations

import json
from pathlib import Path


def summarize(paths: list[Path]) -> str:
    rows = []
    for path in paths:
        data = json.loads(Path(path).read_text())
        metrics = data.get("metrics", {})
        rows.append(
            (
                path.stem,
                data.get("questions", 0),
                metrics.get("recall@10", 0.0),
                metrics.get("ndcg@10", 0.0),
                metrics.get("latency_ms", 0.0),
            )
        )
    header = "| config | n | recall@10 | ndcg@10 | latency_ms |\n|---|---:|---:|---:|---:|\n"
    body = "\n".join(f"| {name} | {n} | {r:.3f} | {nd:.3f} | {lat:.0f} |" for name, n, r, nd, lat in rows)
    return header + body
```
